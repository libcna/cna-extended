5// Ported from MonoGame.Extended's Tilemaps/TilemapLayerCollection.cs. This collection is the
6// canonical owner of every `TilemapLayer` it holds (nothing constructs a layer except via
7// `Add`, and `TilemapGroupLayer`/`TilemapTilesetCollection` only ever *reference* layers/tilesets
8// owned elsewhere) -- `layers_` is an owning `std::vector<std::unique_ptr<TilemapLayer>>`;
9// `layersByName_` is a non-owning `TilemapLayer*` lookup index into the same instances, using
10// this project's own `Dictionary<K,V>` wrapper (fine here since `TilemapLayer*` is a trivially
11// copyable value type, unlike the move-only `TilemapTileData` case in `TilemapTileset.hpp`).

52 template <typename T>
53 [[nodiscard]] T* GetLayer(const std::string& name) const
54 {
55 static_assert(std::is_base_of_v<TilemapLayer, T>, "T must derive from TilemapLayer.");
56 TilemapLayer* layer = nullptr;
57 if (TryGetValue(name, layer))
58 {
59 return dynamic_cast<T*>(layer);
60 }
61 return nullptr;
62 }
63
65 template <typename T>
66 [[nodiscard]] std::vector<T*> GetLayers() const
67 {
68 static_assert(std::is_base_of_v<TilemapLayer, T>, "T must derive from TilemapLayer.");
69 std::vector<T*> result;
70 for (const std::unique_ptr<TilemapLayer>& layer : layers_)
71 {
72 if (auto* typed = dynamic_cast<T*>(layer.get()))
73 {
74 result.push_back(typed);
75 }
76 }
77 return result;
78 }

82

84 [[nodiscard]] auto end() const { return layers_.end(); }
85
