5// Ported from MonoGame.Extended's Serialization/Xml/XmlReaderExtensions.cs. Extension methods ->
6// free functions, matching this project's established convention.
7//
8// Representational note: upstream's `reader.GetAttribute(name)` returns `null` for a missing
9// attribute, distinguishable from `""` for a present-but-empty one; `sharp-runtime`'s
10// `XmlReader::GetAttribute` returns an empty `std::string` for both (verified by reading its
11// implementation). These functions therefore treat an empty attribute value as "missing" -- the
12// closest available behavior given that underlying limitation, not a deliberate simplification.
13//
14// `GetAttributeEnum<T>` is generic over any enum type upstream via `Enum.Parse<T>` (reflection);
15// C++ has no enum reflection. Ported as a template that takes an explicit `parse` callable
16// (`T(*)(const std::string&)` or equivalent) rather than silently only supporting one enum type --
17// this is a genuine, documented representational difference, not a narrowing of upstream's actual
18// generality. No enum type in this codebase has a call site for this yet (its one real upstream
19// caller, `Particles/ParticleEffectSerializer.cs`, is Phase 8 -- not yet ported), so there is
20// nothing to verify this against end-to-end yet; the shape is a direct, deliberate translation of
21// upstream's contract, not a guess.

55 template <typename T>
56 T GetAttributeEnum(const System::Xml::XmlReader& reader, const std::string& attributeName, const std::function<bool(const std::string&, T&)>& parse)
57 {
58 const std::string value = reader.GetAttribute(attributeName);
59 if (value.empty())
60 {
61 throw System::Xml::XmlException("Required attribute '" + attributeName + "' is missing.");
62 }
63
64 T result{};
65 if (!parse(value, result))
66 {
67 throw System::Xml::XmlException("Invalid enum format for attribute '" + attributeName + "'. Got '" + value + "'");
68 }
69 return result;
70 }

73 template <typename T>
74 T GetAttributeEnum(
75 const System::Xml::XmlReader& reader, const std::string& attributeName, T defaultValue,
76 const std::function<bool(const std::string&, T&)>& parse)
77 {
78 const std::string value = reader.GetAttribute(attributeName);
79 if (value.empty())
80 {
81 return defaultValue;
82 }
83
84 T result{};
85 return parse(value, result) ? result : defaultValue;
86 }
