7// GENUINE C#/C++ REPRESENTATIONAL GAP, flagged for the orchestrating session rather than silently
8// resolved (per this project's established practice for hard-to-reverse decisions):
9//
10// Upstream's `ComponentMapper CreateMapperForType(Type type, int componentTypeId)` uses real
11// reflection (`Activator.CreateInstance(typeof(ComponentMapper<>).MakeGenericType(type), ...)`)
12// to construct a `ComponentMapper<T>` for a `T` known only as a runtime `Type` value -- not known
13// at that call site's compile time. This backs the non-generic `this[Type type]` indexer, and
14// upstream's own `ComponentManagerTests.GetMapperForTypeByIndexer` test calls
15// `componentManager[typeof(Transform2)]` as the FIRST access for that type (no prior generic
16// `GetMapper<Transform2>()` call), i.e. it genuinely requires "construct T from nothing but a
17// runtime type token," which C++ cannot do without either (a) a global/pre-registered factory
18// keyed by type, populated in advance for every type that will ever be looked up this way, or
19// (b) true reflection. Neither is available here in a way that preserves upstream's exact
20// on-demand-from-nothing-but-a-Type semantics.
21//
22// Resolution taken (deliberately conservative, easy to revisit): `GetMapper(std::type_index)`
23// below can only return a mapper that was already created via the generic `GetMapper<T>()` path
24// at least once -- it does NOT construct one on first use the way upstream's indexer does. The
25// ported `GetMapperForTypeByIndexerTests` test is adapted to call `GetMapper<T>()` first (matching
26// how every *other* real call site in this ECS module -- `Entity::Attach<T>` etc. -- actually
27// reaches a mapper), then verifies type_index-based lookup returns the same instance. This
28// preserves the verifiable, non-reflective part of upstream's behavior (stable identity/lookup by
29// type) and drops only the "conjure T from a bare runtime token" capability, which no other code
30// in this module actually relies on.
31//
32// A second, smaller gap: `IComponentMapperService` (upstream interface with a generic method,
33// `ComponentMapper<T> GetMapper<T>()`) has no direct translation -- C++ does not allow virtual
34// member function templates, so a generic method cannot be part of a polymorphic interface.
35// `ComponentManager` is the only implementer anywhere in this codebase, so the interface is not
36// separately materialized; `EntitySystem::Initialize` (Systems/EntitySystem.hpp) takes a
37// `ComponentManager&` directly instead of an abstracted interface type.

67 template <typename T>
68 ComponentMapperOf<T>* GetMapper()
69 {
70 const int componentTypeId = GetComponentTypeId(std::type_index(typeid(T)));
71
72 ComponentMapper* existing = componentMappers_[static_cast<std::size_t>(componentTypeId)];
73 if (existing != nullptr)
74 {
75 return static_cast<ComponentMapperOf<T>*>(existing);
76 }
77
78 return CreateMapperForType<T>(componentTypeId);
79 }

98 int GetComponentTypeId(std::type_index type)
99 {
100 const auto it = componentTypes_.find(type);
101 if (it != componentTypes_.end())
102 {
103 return it->second;
104 }
105
106 const auto id = static_cast<int>(componentTypes_.size());
107 componentTypes_.emplace(type, id);
108 return id;
109 }
110
112 [[nodiscard]] ComponentBits CreateComponentBits(int entityId) const
113 {
114 ComponentBits componentBits;
115
116 for (std::size_t componentId = 0; componentId < componentMappers_.getCountProperty(); ++componentId)
117 {
118 const ComponentMapper* mapper = componentMappers_[componentId];
119 componentBits[static_cast<int>(componentId)] = mapper != nullptr && mapper->Has(entityId);
120 }
121
122 return componentBits;
123 }
124
126 void Destroy(int entityId)
127 {
128 for (ComponentMapper* mapper : componentMappers_)
129 {
130 if (mapper != nullptr)
131 {
132 mapper->Delete(entityId);
133 }
134 }
135 }
136
137 void Update(const GameTime& gameTime) override { (void)gameTime; }
138
140 [[nodiscard]] const Collections::Bag<ComponentMapper*>& getComponentMappersProperty() const { return componentMappers_; }
141
142 [[nodiscard]] auto begin() const { return componentMappers_.begin(); }
143 [[nodiscard]] auto end() const { return componentMappers_.end(); }
144
145 private:
146 template <typename T>
147 ComponentMapperOf<T>* CreateMapperForType(int componentTypeId)
148 {
149 if (componentTypeId >= MaxComponentTypes)
150 {
151 throw std::invalid_argument("Component type limit exceeded. Maximum of 256 component types are supported.");
152 }
153
154 auto mapper = std::make_unique<ComponentMapperOf<T>>(componentTypeId, &ComponentsChanged);
155 ComponentMapperOf<T>* result = mapper.get();
156 ownedMappers_.push_back(std::move(mapper));
157 componentMappers_.Set(static_cast<std::size_t>(componentTypeId), result);
158 return result;
159 }
160
161 std::vector<std::unique_ptr<ComponentMapper>> ownedMappers_;
162 Collections::Bag<ComponentMapper*> componentMappers_;
163 std::unordered_map<std::type_index, int> componentTypes_;
164 };
165}
