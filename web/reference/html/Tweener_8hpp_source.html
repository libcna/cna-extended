5// Ported from MonoGame.Extended's Tweening/Tweener.cs. Manages a collection of active tween
6// animations and drives their updates each frame. Create animations using TweenTo(), then call
7// Update() once per frame.
8//
9// *** GENUINE C++ REDESIGN, NOT A LITERAL PORT -- see TweenMember.hpp/Tween.hpp for the full
10// rationale ***. TweenTo's calling convention changes from a lambda member-selector
11// (`tweener.TweenTo(target, x => x.Position, endValue, duration)`) to C++ pointer-to-member
12// syntax (`tweener.TweenTo(&target, &TargetType::Position, endValue, duration)`).
13//
14// Return type: upstream's `TweenTo<TTarget,TMember>` returns `Tween<TMember>` (the GENERIC base,
15// not the concrete LinearTween<T>/ColorTween subtype it actually constructs) -- ported faithfully
16// as `TypedTween<TTarget,TMember>*` (the renamed equivalent, see TypedTween.hpp), not
17// `LinearTween<TTarget,TMember>*`. This also cleanly unifies the Color-vs-non-Color dispatch
18// below into a single template (both branches return the same TypedTween<TTarget,TMember>*),
19// since C++ doesn't allow two identically-named template overloads that would otherwise be
20// ambiguous for TMember=Color.
21//
22// Upstream's `switch (toValue) { case Color: ... }` runtime dispatch -> `if constexpr
23// (std::is_same_v<TMember, Color>)` compile-time dispatch: since TMember is a template parameter
24// here (unlike upstream's boxed runtime value), the Color-vs-linear choice is already known at
25// compile time, so a runtime type switch is unnecessary -- another simplification the stronger
26// C++ type system affords for free.
27//
28// Upstream's `Activator.CreateInstance(typeof(TTween), BindingFlags.Public|BindingFlags.NonPublic,
29// ...)` (reflection-based construction of an arbitrary TTween subtype, needed because upstream's
30// TweenTo<TTarget,TMember,TTween> lets the CALLER specify a custom TTween type by reflection) has
31// no equivalent here and isn't needed: TweenTo always constructs exactly LinearTween or ColorTween
32// directly via std::make_unique, since those are the only two concrete Tween kinds this port
33// supports (matching what upstream's own public TweenTo<TTarget,TMember> overload -- the one
34// without an explicit TTween type parameter -- actually does). Upstream's separate 3-type-parameter
35// TweenTo<TTarget,TMember,TTween> overload (letting callers plug in a wholly custom TTween
36// subclass) is not ported -- it exists upstream purely to make the reflection-based construction
37// path extensible to third-party Tween subtypes, which has no analogue once construction is a
38// direct, statically-typed std::make_unique call.
39//
40// Upstream's member cache (`Dictionary<TweenMemberKey, TweenMember> _memberCache`, keyed by
41// (Target, MemberName) using reflection-based names) is ELIMINATED: it existed only to amortize
42// the cost of reflection-based TweenMember construction, which no longer exists -- constructing a
43// TweenMember<TTarget,TMember> is a trivial, cheap pointer-pair copy with nothing to cache.
44//
45// FindTween/the "cancel an existing tween on the same member before starting a new one" logic in
46// TweenTo both use `dynamic_cast<TypedTween<TTarget,TMember>*>` to narrow each stored `Tween*` to
47// the concrete type being searched for, then compare (target pointer, pointer-to-member value)
48// directly with `==` -- both are genuinely comparable with `==` in standard C++, uniquely
49// identifying a bound member without needing any name string (unlike upstream's
50// `t.Target == target && t.MemberName == memberName` string comparison, which existed only
51// because reflection gave upstream a member NAME but no cheaper identity to compare).
52//
53// Header-only for the template TweenTo/FindTween methods (must be visible to every calling
54// translation unit, since TTarget/TMember are supplied by the caller); non-template methods
55// (Update/CancelAll/CancelAndCompleteAll/Dispose) are declared here and defined in Tweener.cpp.

148 template <typename TTarget, typename TMember>
149 TypedTween<TTarget, TMember>* TweenTo(TTarget* target, TMember TTarget::*member, TMember toValue, float duration, float delay = 0.0f)
150 {
151 if (TypedTween<TTarget, TMember>* existing = FindTween(target, member))
152 {
153 existing->Cancel();
154 }
155
156 TweenMember<TTarget, TMember> tweenMember(target, member);
157 std::unique_ptr<TypedTween<TTarget, TMember>> tween;
158 if constexpr (std::is_same_v<TMember, Color>)
159 {
160 tween = std::make_unique<ColorTween<TTarget>>(tweenMember, duration, delay, toValue);
161 }
162 else
163 {
164 tween = std::make_unique<LinearTween<TTarget, TMember>>(tweenMember, duration, delay, toValue);
165 }
166
167 TypedTween<TTarget, TMember>* result = tween.get();
168 activeTweens_.push_back(std::move(tween));
169 return result;
170 }

176 template <typename TTarget, typename TMember>
177 [[nodiscard]] TypedTween<TTarget, TMember>* FindTween(TTarget* target, TMember TTarget::*member) const
178 {
179 for (const auto& tween : activeTweens_)
180 {
181 if (auto* typed = dynamic_cast<TypedTween<TTarget, TMember>*>(tween.get()))
182 {
183 if (typed->getMemberProperty().getTargetProperty() == target && typed->getMemberProperty().getMemberPointerProperty() == member)
184 {
185 return typed;
186 }
187 }
188 }
189 return nullptr;
190 }
