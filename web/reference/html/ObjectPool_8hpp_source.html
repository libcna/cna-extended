5// Ported from MonoGame.Extended's Collections/ObjectPool.cs: an object pool for IPoolable-
6// implementing types, tracking free items in a circular-buffer Deque (O(1) operations) and
7// in-use items in an intrusive doubly-linked list (via IPoolable::NextNode/PreviousNode), so the
8// whole in-use set can be iterated without a separate allocation.
9//
10// C#'s `where T : class, IPoolable` reference-type + interface constraint maps to `T` denoting
11// the pooled class itself (must derive from IPoolable, enforced via static_assert), with the
12// pool working in T* throughout -- matching how ObjectPool<T> in upstream already always holds
13// references, and how Pool<T>/GameComponentCollectionExtensions.hpp translate C# reference-type
14// generic parameters elsewhere in this project. `Func<T>` -> `std::function<T*()>`.
15// `event Action<T> ItemUsed`/`ItemReturned` -> `System::MulticastAction<T*>` (a true C# multicast
16// event, unlike EventHandler<TEventArgs>'s (sender, args) shape -- MulticastAction<Args...> was
17// added to sharp-runtime earlier in this project specifically for this kind of need).
18//
19// Everywhere upstream does an unchecked `(T)node.NextNode`/`(T)item` cast (trusting that only
20// T-implementing instances are ever linked into a given ObjectPool<T>'s own list), this uses
21// `static_cast<T*>` -- well-defined given that same trusted invariant, matching upstream's own
22// lack of a runtime type check at these points.
23//
24// *** SEVERE UPSTREAM BUG, PRESERVED FOR FIDELITY -- causes a genuine infinite loop, not just a
25// wrong value; confirmed by hand-tracing the exact C# source byte-for-byte, not assumed ***
26// `CreateObject()` unconditionally sets `_tailNode = item` as its LAST step, before returning to
27// `New()`, which then immediately calls `Use(item)` on that same item. `Use()`'s very first
28// structural decision is `if (_tailNode is null)` -- intended to detect "is this the pool's very
29// first-ever node" and, if so, leave its PreviousNode/NextNode as null (a proper list head/tail).
30// But because `CreateObject()` already set `_tailNode = item` moments earlier, this check is
31// FALSE for every freshly-created item, without exception -- the "is this the first node" branch
32// is unreachable dead code. `Use()` instead always takes the `else` branch, which sets
33// `item->PreviousNode = _tailNode` and `_tailNode->NextNode = item` -- but since `_tailNode` IS
34// `item` at this point, this makes the node point to ITSELF: `item.PreviousNode == item` and
35// `item.NextNode == item`.
36// For a pool where this is the very first (and, so far, only) item ever created, this means
37// `headNode_ == tailNode_ == item`, with `item->NextNode == item` (a self-loop). `GetEnumerator`
38// (`while (node != null) { yield return node; node = node.NextNode; }`) then NEVER terminates,
39// because `node.NextNode` is `node` itself -- an infinite loop on the single most basic usage
40// pattern of this class (create one pooled object, then enumerate the pool). Confirmed by hand-
41// tracing this exact single-item scenario line-by-line against the C# source (not assumed).
42// More generally: since `_tailNode` is always exactly the item just passed to `Use()` at the
43// moment of that call, EVERY freshly-created item's NextNode briefly self-references, and it is
44// only "fixed" retroactively -- either because a later `CreateObject()` call points the *former*
45// tail's NextNode at the newly-created item instead (see `CreateObject()`'s
46// `if (_tailNode != null) _tailNode.NextNode = item;`), or because `Return()` unconditionally
47// resets the CURRENT tail's NextNode to null as its last bookkeeping step
48// (`if (_tailNode != null) _tailNode.NextNode = null;`). So enumerating a pool where every item
49// was obtained purely via `New()` and NONE has ever been `Return()`ed yet will always hang on the
50// current tail node, not just in the single-item case -- this port only directly proves and
51// tests the single-item case below (the simplest, most certainly-reproducible instance), rather
52// than claiming to have exhaustively verified every possible sequence of New()/Return() calls.
53// Reproduced exactly below via a faithful line-by-line translation, NOT fixed -- see
54// `ObjectPoolTests.cpp`'s regression tests, which assert the self-referencing pointers directly
55// via `getNextNodeProperty()`/`getPreviousNodeProperty()` rather than by iterating the pool
56// (iterating in that exact scenario would hang the test process forever, since it is the bug
57// itself).
58//

109 explicit ObjectPool(std::function<T*()> instantiationFunction, const intcs capacity = 16,
110 const ObjectPoolIsFullPolicy isFullPolicy = ObjectPoolIsFullPolicy::ReturnNull)
111 : freeItems_(capacity),
112 instantiationFunction_(std::move(instantiationFunction)),
113 isFullPolicy_(isFullPolicy),

116 returnToPoolDelegate_ = [this](IPoolable* item) { Return(item); };
117 }

120 [[nodiscard]] intcs getCapacityProperty() const { return capacity_; }
121 [[nodiscard]] intcs getTotalCountProperty() const { return totalCount_; }
122 [[nodiscard]] intcs getAvailableCountProperty() const { return freeItems_.getCountProperty(); }
123 [[nodiscard]] intcs getInUseCountProperty() const { return totalCount_ - getAvailableCountProperty(); }

130 T* New()
131 {
132 T* poolable = nullptr;
133 if (!freeItems_.RemoveFromFront(poolable))
134 {
135 if (totalCount_ <= capacity_)
136 {
137 poolable = CreateObject();
138 }
139 else
140 {
141 switch (isFullPolicy_)
142 {
143 case ObjectPoolIsFullPolicy::ReturnNull:
144 return nullptr;
145 case ObjectPoolIsFullPolicy::IncreaseSize:
146 capacity_++;
147 poolable = CreateObject();
148 break;
149 case ObjectPoolIsFullPolicy::KillExisting:
150 if (headNode_ == nullptr)
151 {
152 return nullptr;
153 }
154 {
155 T* newHeadNode = static_cast<T*>(headNode_->getNextNodeProperty());
156 headNode_->Return();
157 freeItems_.RemoveFromBack(poolable);
158 headNode_ = newHeadNode;
159 }
160 break;
161 default:
162 throw std::logic_error("ObjectPool<T>::New: unrecognized ObjectPoolIsFullPolicy");
163 }
164 }
165 }
166
167 Use(poolable);
168 return poolable;
169 }

196 private:
197 T* CreateObject()
198 {
199 totalCount_++;
200 T* item = instantiationFunction_();
201 if (item == nullptr)
202 {
203 throw std::runtime_error("The created pooled object is null.");
204 }
205 item->setPreviousNodeProperty(tailNode_);
206 item->setNextNodeProperty(nullptr);
207 if (headNode_ == nullptr)
208 {
209 headNode_ = item;
210 }
211 if (tailNode_ != nullptr)
212 {
213 tailNode_->setNextNodeProperty(item);
214 }
215 tailNode_ = item;
216 return item;
217 }

219 void Return(IPoolable* item)
220 {
221 T* poolable = static_cast<T*>(item);
222
223 T* previousNode = static_cast<T*>(item->getPreviousNodeProperty());
224 T* nextNode = static_cast<T*>(item->getNextNodeProperty());
225
226 if (previousNode != nullptr)
227 {
228 previousNode->setNextNodeProperty(nextNode);
229 }
230 if (nextNode != nullptr)
231 {
232 nextNode->setPreviousNodeProperty(previousNode);
233 }
234
235 if (item == headNode_)
236 {
237 headNode_ = nextNode;
238 }
239 if (item == tailNode_)
240 {
241 tailNode_ = previousNode;
242 }
243
244 if (tailNode_ != nullptr)
245 {
246 tailNode_->setNextNodeProperty(nullptr);
247 }
248
249 freeItems_.AddToBack(poolable);
250
251 ItemReturned(poolable);
252 }
253
254 void Use(T* item)
255 {
256 item->Initialize(returnToPoolDelegate_);
257 item->setNextNodeProperty(nullptr);
258 // See this file's header comment (SEVERE UPSTREAM BUG): CreateObject() always sets
259 // tailNode_ = item as its last step before calling Use(item) on that same item, so
260 // this check is FALSE for every freshly-created item without exception -- the
261 // "is this the pool's first node" branch below is unreachable dead code in that
262 // path. The else branch then links item to itself (PreviousNode == item,
263 // NextNode == item), which causes GetEnumerator() to loop forever once this is the
264 // current tail. Preserved exactly, not fixed.
265 if (tailNode_ == nullptr)
266 {
267 headNode_ = item;
268 tailNode_ = item;
269 item->setPreviousNodeProperty(nullptr);
270 }
271 else
272 {
273 item->setPreviousNodeProperty(tailNode_);
274 tailNode_->setNextNodeProperty(item);
275 tailNode_ = item;
276 }
277
278 ItemUsed(item);
279 }
