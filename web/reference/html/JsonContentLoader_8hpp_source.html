26 class JsonContentLoader
27 {
28 public:
29 template <typename T>
30 [[nodiscard]] T Load(ContentManager& contentManager, const std::string& path) const
31 {
32 const std::unique_ptr<System::IO::Stream> stream = CNA::Extended::Content::OpenStream(contentManager, path);
33 const System::Text::Json::JsonSerializerOptions options = GetOptions(contentManager, path);
34
35 std::vector<SharpRuntime::bytecs> buffer(static_cast<std::size_t>(stream->getLengthProperty()));
36 SharpRuntime::intcs totalRead = 0;
37 while (totalRead < stream->getLengthProperty())
38 {
39 const SharpRuntime::intcs read =
40 stream->Read(buffer.data(), totalRead, stream->getLengthProperty() - totalRead);
41 if (read == 0)
42 {
43 break;
44 }
45 totalRead += read;
46 }
47
48 const std::string json(buffer.begin(), buffer.begin() + totalRead);
49 return System::Text::Json::JsonSerializer::Deserialize<T>(json, options);
50 }
51 };
52}
