5// Ported from MonoGame.Extended's Serialization/Json/IntervalJsonConverter.cs (a generic
6// `JsonConverter<Interval<T>>`, upstream constrained to `T : IComparable<T>`). `Interval<T>` is
7// already a C++ template (Interval.hpp); this ports as a *partial* nlohmann::adl_serializer
8// specialization over `T`, header-only since it's a template -- see ColorJsonConverter.hpp's
9// header comment for the overall adl_serializer design rationale shared by this module.

18namespace nlohmann
19{
20 template <typename T>
21 struct adl_serializer<CNA::Extended::Interval<T>>
22 {
23 static void to_json(nlohmann::ordered_json& j, const CNA::Extended::Interval<T>& value)
24 {
25 j = nlohmann::ordered_json::array({value.getMinProperty(), value.getMaxProperty()});
26 }
27
28 static CNA::Extended::Interval<T> from_json(const nlohmann::ordered_json& j)
29 {
30 const std::vector<T> values = CNA::Extended::Serialization::Json::ReadAsMultiDimensional<T>(j);
31
32 if (values.size() == 2)
33 {
34 return values[0] < values[1] ? CNA::Extended::Interval<T>(values[0], values[1])
35 : CNA::Extended::Interval<T>(values[1], values[0]);
36 }
37
38 if (values.size() == 1)
39 {
40 return CNA::Extended::Interval<T>(values[0], values[0]);
41 }
42
43 throw std::invalid_argument("Invalid interval");
44 }
45 };
