5// Ported from MonoGame.Extended's Tilemaps/TilemapLayers/TilemapObjectLayer.cs. Unlike
6// `TilemapGroupLayer`'s child-layer references or `TilemapTilesetCollection`'s tileset
7// references, nothing else in this data model references a specific placed `TilemapObject`
8// instance once it's added to an object layer -- this layer is each object's sole, canonical
9// owner, so `_objects` is translated as an owning `std::vector<std::unique_ptr<TilemapObject>>`
10// (contrast with `TilemapTileData::CollisionObjects`, which is a *different*, also-owning
11// collection of per-tile collision shapes -- the two are unrelated despite the shared element
12// type). `IEnumerable<T> GetObjectsInRegion`/`GetObjects<T>` (C# generator methods using `yield
13// return`) become eagerly-evaluated `std::vector` results, matching this project's established
14// convention for LINQ/generator-style query methods.

56 template <typename T>
57 [[nodiscard]] std::vector<T*> GetObjects() const
58 {
59 static_assert(std::is_base_of_v<TilemapObject, T>, "T must derive from TilemapObject.");
60 std::vector<T*> result;
61 for (const std::unique_ptr<TilemapObject>& obj : objects_)
62 {
63 if (auto* typedObj = dynamic_cast<T*>(obj.get()))
64 {
65 result.push_back(typedObj);
66 }
67 }
68 return result;
69 }
