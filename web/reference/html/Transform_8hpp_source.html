40 enum class TransformFlags : std::uint8_t
41 {
42 None = 0,
43 WorldMatrixIsDirty = 1 << 0,
44 LocalMatrixIsDirty = 1 << 1,
45 All = WorldMatrixIsDirty | LocalMatrixIsDirty
46 };
47
48 constexpr TransformFlags operator|(const TransformFlags a, const TransformFlags b)
49 {
50 return static_cast<TransformFlags>(static_cast<std::uint8_t>(a) | static_cast<std::uint8_t>(b));
51 }
52
53 constexpr TransformFlags operator&(const TransformFlags a, const TransformFlags b)
54 {
55 return static_cast<TransformFlags>(static_cast<std::uint8_t>(a) & static_cast<std::uint8_t>(b));
56 }
57
58 constexpr TransformFlags operator~(const TransformFlags a)
59 {
60 return static_cast<TransformFlags>(static_cast<std::uint8_t>(~static_cast<std::uint8_t>(a)));
61 }

176 using AncestorSubscription = std::pair<BaseTransform*, typename System::MulticastAction<>::Token>;
177
178 void onParentChanged(BaseTransform* oldParent, BaseTransform* newParent)
179 {
180 for (BaseTransform* p = oldParent; p != nullptr; p = p->parent_)
181 {
182 unsubscribeFrom(p);
183 }
184
185 for (BaseTransform* p = newParent; p != nullptr; p = p->parent_)
186 {
187 subscribeTo(p);
188 }
189
190 flags_ |= TransformFlags::All;
191 }
192
193 void subscribeTo(BaseTransform* ancestor)
194 {
195 const auto token = ancestor->TransformBecameDirty.Add([this]() { parentOnTransformBecameDirty(); });
196 ancestorSubscriptions_.emplace_back(ancestor, token);
197 }
198
199 void unsubscribeFrom(BaseTransform* ancestor)
200 {
201 const auto it = std::find_if(ancestorSubscriptions_.begin(), ancestorSubscriptions_.end(),
202 [ancestor](const AncestorSubscription& entry) { return entry.first == ancestor; });
203 if (it != ancestorSubscriptions_.end())
204 {
205 ancestor->TransformBecameDirty.Remove(it->second);
206 ancestorSubscriptions_.erase(it);
207 }
208 }
209
210 void parentOnTransformBecameDirty()
211 {
212 flags_ |= TransformFlags::All;
213 }
214
215 void recalculateWorldMatrixIfNecessary()
216 {
217 if ((flags_ & TransformFlags::WorldMatrixIsDirty) == TransformFlags::None)
218 {
219 return;
220 }
221
222 recalculateLocalMatrixIfNecessary();
223 RecalculateWorldMatrix(localMatrix_, worldMatrix_);
224
225 flags_ &= ~TransformFlags::WorldMatrixIsDirty;
226 TranformUpdated();
227 }
228
229 void recalculateLocalMatrixIfNecessary()
230 {
231 if ((flags_ & TransformFlags::LocalMatrixIsDirty) == TransformFlags::None)
232 {
233 return;
234 }
235
236 RecalculateLocalMatrix(localMatrix_);
237
238 flags_ &= ~TransformFlags::LocalMatrixIsDirty;
239 WorldMatrixBecameDirty();
240 }
241
242 TransformFlags flags_ = TransformFlags::All; // dirty flags, set all dirty flags when created
243 TMatrix localMatrix_{}; // model space to local space
244 TMatrix worldMatrix_{}; // local space to world space
245 BaseTransform* parent_ = nullptr;
246 std::vector<AncestorSubscription> ancestorSubscriptions_;
247 };
