5// Ported from MonoGame.Extended's Tweening/ColorTween.cs: a specialized tween for
6// Microsoft::Xna::Framework::Color, using Color::Lerp for interpolation since a packed color
7// doesn't support the plain arithmetic operators LinearTween<TTarget,TMember> relies on. Templated
8// only on TTarget here (unlike upstream's non-generic `ColorTween : Tween<Color>`), since the
9// value type is fixed to Color -- Tweener::TweenTo dispatches to this specialization whenever
10// TMember is Color, matching upstream's own `switch (toValue) { case Color: ... }` dispatch (see
11// Tweener.hpp).
12//
13// Header-only: a template, matching this project's established convention for generic C# types.
14#pragma once
15

24 template <typename TTarget>

31 }

33 protected:
34 void Interpolate(float n) override

37 this->getMemberProperty().setValueProperty(value);
38 }
