12// This is an internal implementation-detail collection used only by Triangulator (upstream's own
13// doc comment on Triangulator: "This is the sole public class in the entire library; all other
14// classes/structures are intended as internal-only objects"). Upstream implements it as
15// `class CyclicalList<T> : List<T>` with a `new T this[int]` cyclical indexer and `new RemoveAt`
16// hiding the base List<T> members via C#'s `new` keyword. C++ has no equivalent method-hiding
17// mechanism worth replicating for an internal type with no public API surface, so this composes
18// (rather than inherits) sharp-runtime's System::Collections::Generic::List<T> instead, per this
19// project's "reuse sharp-runtime, don't re-roll" convention -- List<T> already provides exactly
20// the operations Triangulator.cs actually uses (Add, Contains, Remove, IndexOf, Clear, begin/end);
21// only the cyclical operator[]/RemoveAt are added here. Header-only template, matching the
22// precedent set by Interval<T> (include/CNA/Extended/Interval.hpp) for header-only generic types
23// in this project -- even though in practice this is only ever instantiated with Vertex.

76 [[nodiscard]] intcs WrapIndex(intcs index) const
77 {
78 const intcs count = getCountProperty();
79 while (index < 0)
80 {
81 index = count + index;
82 }
83 if (index >= count)
84 {
85 index %= count;
86 }
87 return index;
88 }
