5// Ported from MonoGame.Extended's Serialization/Json/Utf8JsonReaderExtensions.cs. Upstream reads
6// from a streaming `Utf8JsonReader` (System.Text.Json's token-by-token reader); sharp-runtime's
7// `JsonSerializer` is DOM-based (built on nlohmann's parsed `ordered_json` tree, not a streaming
8// reader -- see JsonSerializer.hpp's own header comment), so this operates on an already-parsed
9// `nlohmann::ordered_json` value instead of a reader reference. Same three supported shapes as
10// upstream (JSON array, delimited string, single number), same fallback order.
11//
12// `ReadAsDelimitedString`'s upstream dispatch table is a closed `Dictionary<Type, Func<string,
13// object>>` covering exactly three types (`int`, `float`, `HslColor`) -- not real reflection.
14// Translated as explicit function-template specializations below (`ParseDelimitedElement<T>`),
15// matching that same closed set; adding a fourth type upstream never happened either.

51 template <typename T>
52 std::vector<T> ReadAsDelimitedString(const std::string& value)
53 {
54 std::vector<T> result;
55 if (value.empty())
56 {
57 return result;
58 }
59
60 std::istringstream stream(value);
61 std::string token;
62 while (stream >> token)
63 {
64 result.push_back(ParseDelimitedElement<T>(token));
65 }
66 return result;
67 }

73 template <typename T>
74 std::vector<T> ReadAsMultiDimensional(const nlohmann::ordered_json& j)
75 {
76 if (j.is_array())
77 {
78 std::vector<T> result;
79 result.reserve(j.size());
80 for (const auto& element : j)
81 {
82 result.push_back(element.template get<T>());
83 }
84 return result;
85 }
86
87 if (j.is_string())
88 {
89 return ReadAsDelimitedString<T>(j.template get<std::string>());
90 }
91
92 if (j.is_number())
93 {
94 return std::vector<T>{j.template get<T>()};
95 }
96
97 throw std::invalid_argument("is not currently supported in the multi-dimensional parser");
98 }
