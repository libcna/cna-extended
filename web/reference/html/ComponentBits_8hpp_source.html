5// Ported from MonoGame.Extended's ECS/ComponentBits.cs: a fixed 256-bit field (4 segments of 64
6// bits) for tracking which component types are present on an entity. Upstream implements its
7// indexer via `Unsafe.Add(ref _bits0, segmentIndex)` -- raw pointer arithmetic treating four
8// separate `ulong` fields as a contiguous 4-element array, done purely to avoid a real array's
9// bounds-check overhead. Translated as a plain `std::array<std::uint64_t, 4>` indexed normally;
10// this project has no equivalent need to dodge C#-specific struct-layout/field-address tricks,
11// and a real array is both simpler and no less safe than the upstream reinterpretation.

129 private:
130 [[nodiscard]] bool Get(int bitIndex) const
131 {
132 CheckIndex(bitIndex);
133 const auto segmentIndex = static_cast<std::size_t>(bitIndex >> 6);
134 const int bitPosition = bitIndex & 63;
135 const std::uint64_t mask = std::uint64_t{1} << bitPosition;
136 return (bits_[segmentIndex] & mask) != 0;
137 }
138
139 void Set(int bitIndex, bool value)
140 {
141 CheckIndex(bitIndex);
142 const auto segmentIndex = static_cast<std::size_t>(bitIndex >> 6);
143 const int bitPosition = bitIndex & 63;
144 const std::uint64_t mask = std::uint64_t{1} << bitPosition;
145 if (value)
146 {
147 bits_[segmentIndex] |= mask;
148 }
149 else
150 {
151 bits_[segmentIndex] &= ~mask;
152 }
153 }
154
155 static void CheckIndex(int bitIndex)
156 {
157 if (bitIndex < 0 || bitIndex >= MaxBits)
158 {
159 throw std::out_of_range("bitIndex must be between 0 and 255.");
160 }
161 }
162
163 std::array<std::uint64_t, SegmentCount> bits_{};
164 };
